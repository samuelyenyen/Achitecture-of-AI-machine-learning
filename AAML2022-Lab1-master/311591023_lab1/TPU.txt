module TPU(input clk, rst_n, in_valid,
           input [7:0]       K, M, N,
           input [31:0]      A_data_out, B_data_out,
           input [127:0]     C_data_out,
           output            A_wr_en, B_wr_en, C_wr_en,
           output reg [15:0]     A_index, B_index, C_index,
           output reg [31:0]     A_data_in, B_data_in, 
           output reg [127:0]    C_data_in,
           output reg         busy);

//--------------------------state machine-----------------------------------------------------
reg [2:0] curr_state, next_state; parameter IDLE = 3'b000, CAL = 3'b001, WR = 3'b011, RESETPE = 3'b010;
//state machine
always @(posedge clk or negedge rst_n) curr_state <= (!rst_n)? IDLE: next_state;      
always @(*)begin case (curr_state) IDLE   : next_state = (in_valid)? CAL: IDLE;
                                   CAL    : next_state = (pe_end==1)? WR: CAL;
                                   WR     : next_state = (wr_end_r==1 & cal_end==1)? IDLE: (wr_end_r==1)? RESETPE:  WR;
                                   RESETPE: next_state = CAL;
                                   default : next_state = IDLE ; endcase end
//--------------------------CAL-----------------------------------------------------
wire pe_end, wr_end, cal_end; reg [7:0] K_r, M_r, N_r; reg [31:0] PE [0:3] [0:3];integer i,j;reg have_zero;reg wr_end_r;
wire [6:0] n_A,n_B; reg [15:0] C_index_r; wire [12:0] B_wn,A_wn; wire [12:0] cal_pos; reg [31:0] cal_count; reg [1:0] wr_count;
always @(posedge clk or negedge rst_n) busy <= (!rst_n)? 0: (next_state==IDLE)? 0: 1;
always @(posedge clk) A_index <= (next_state==CAL)? A_index+1: (next_state==IDLE)? 0: (next_state==WR)? n_A*K_r: A_index;
always @(posedge clk) B_index <= (next_state==CAL)? B_index+1: (next_state==IDLE)? 0: (next_state==WR)? n_B*K_r: B_index;
always @(posedge clk) C_index_r <= (next_state==WR)? C_index_r+1: (next_state==IDLE)? 0: C_index_r;
always @(posedge clk) cal_count <= (next_state==IDLE)? 0: (next_state==CAL)? cal_count+1: cal_count;
always @(posedge clk) wr_count <= (next_state==WR)? wr_count+1: 0;
assign cal_pos = cal_count/K_r;
always @(posedge clk) have_zero <= (next_state==CAL)? ((n_A+1==A_wn)? 1:0): have_zero;
assign n_A = cal_pos%A_wn;
assign n_B = cal_pos/A_wn;
assign pe_end = (A_index%K_r == 0)? 1:0;
assign cal_end = (cal_pos==B_wn*A_wn)? 1:0;
//assign wr_end = (wr_count == 3)? 1:0;
always @(posedge clk) wr_end_r<=wr_end;
assign wr_end = (M_r%4==0)?((wr_count ==3)? 1: 0): (have_zero)? ((wr_count==M_r%4-1)? 1: 0): ((wr_count ==3)? 1: 0);
assign C_wr_en = (curr_state==WR)? 1: 0;
assign B_wn = (N_r-1)/4+1;
assign A_wn = (M_r-1)/4+1;
always @(posedge clk) C_index <= C_index_r;
always @(posedge clk) begin 
    if (next_state==WR) begin
        C_data_in[127:96] <= PE[wr_count][0];
        C_data_in[95:64] <= PE[wr_count][1];
        C_data_in[63:32] <= PE[wr_count][2];
        C_data_in[31:0] <= PE[wr_count][3];
    end
end
assign A_wr_en = 0;
assign B_wr_en = 0;
always @(posedge clk) A_data_in <= 0;
always @(posedge clk) B_data_in <= 0;

always @(posedge clk) begin
    if(next_state==IDLE | next_state==RESETPE) begin
        for(i=0;i<4;i=i+1) 
            for(j=0;j<4;j=j+1)
                PE[i][j]<=32'd0;
    end
    else if (next_state==CAL) begin
        PE[0][0] <= PE[0][0]+A_data_out[31:24]*B_data_out[31:24];
        PE[0][1] <= PE[0][1]+A_data_out[31:24]*B_data_out[23:16];
        PE[0][2] <= PE[0][2]+A_data_out[31:24]*B_data_out[15:8];
        PE[0][3] <= PE[0][3]+A_data_out[31:24]*B_data_out[7:0];
        PE[1][0] <= PE[1][0]+A_data_out[23:16]*B_data_out[31:24];
        PE[1][1] <= PE[1][1]+A_data_out[23:16]*B_data_out[23:16];
        PE[1][2] <= PE[1][2]+A_data_out[23:16]*B_data_out[15:8];
        PE[1][3] <= PE[1][3]+A_data_out[23:16]*B_data_out[7:0];
        PE[2][0] <= PE[2][0]+A_data_out[15:8]*B_data_out[31:24];
        PE[2][1] <= PE[2][1]+A_data_out[15:8]*B_data_out[23:16];
        PE[2][2] <= PE[2][2]+A_data_out[15:8]*B_data_out[15:8];
        PE[2][3] <= PE[2][3]+A_data_out[15:8]*B_data_out[7:0];
        PE[3][0] <= PE[3][0]+A_data_out[7:0]*B_data_out[31:24];
        PE[3][1] <= PE[3][1]+A_data_out[7:0]*B_data_out[23:16];
        PE[3][2] <= PE[3][2]+A_data_out[7:0]*B_data_out[15:8];
        PE[3][3] <= PE[3][3]+A_data_out[7:0]*B_data_out[7:0];
    end
end

always @(posedge clk) K_r <= (in_valid)? K: K_r;
always @(posedge clk) M_r <= (in_valid)? M: M_r;
always @(posedge clk) N_r <= (in_valid)? N: N_r;

endmodule
